
# Task 1: Find the total price of each of the first 5 orders, ordered by order_id.

SELECT 
    orders.order_id, 
    SUM(products.price * line_items.quantity) AS total_price 
FROM orders 

JOIN line_items 
    ON orders.order_id = line_items.order_id 
JOIN products 
    ON line_items.product_id = products.product_id 
GROUP BY orders.order_id 
ORDER BY orders.order_id 
LIMIT 5; 


# Task 2: Understanding Subqueries - Find the average total order price for each customer. 

SELECT 
    c.customer_name, 
    AVG(t.order_total) AS average_order_price 
FROM customers AS c 

JOIN ( 
    SELECT 
        o.customer_id, 
        SUM(p.price * li.quantity) AS order_total 
    FROM orders AS o 
    JOIN line_items AS li 
        ON o.order_id = li.order_id 
    JOIN products AS p 
        ON li.product_id = p.product_id 
    GROUP BY o.customer_id, o.order_id 
) AS t 
    ON c.customer_id = t.customer_id 
GROUP BY c.customer_id, c.customer_name 
ORDER BY c.customer_name;

# Task 4: Aggregation with Having - Find employees with more than 5 orders. Return the first_name, the last_name, and the count of orders as order_count.

SELECT 
    e.first_name, 
    e.last_name, 
    COUNT(o.order_id) AS order_count 
FROM employees AS e 

JOIN orders AS o 
    ON e.employee_id = o.employee_id 
GROUP BY 
    e.employee_id, 
    e.first_name, 
    e.last_name 
HAVING COUNT(o.order_id) > 5 
ORDER BY e.last_name;
